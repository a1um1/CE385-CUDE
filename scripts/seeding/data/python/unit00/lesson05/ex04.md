  def anagram(a: str, b: str) -> bool:

testerCode: |
  a = input()
  b = input()
  print(anagram(a, b))
testCase:
  - input: "listen\nsilent"
    output: "True"
    public: true
  - input: "hello\nhallo"
    output: "False"
    public: true
  - input: "Clint Eastwood\nold west action"
    output: "True"
  - input: "rat\ncar"
    output: "False"
  - input: "a\na"
    output: "True"

## Prototype

```
anagram(a: str, b: str) -> bool
```

## Expected Output

```

> listen
> silent
True
```
